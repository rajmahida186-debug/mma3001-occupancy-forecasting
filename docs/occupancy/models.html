occupancy.models

Forecasting models and HVAC schedules.

The machine-learning model, GradientBoostingForecaster, is compared against two simple reference solutions:

  • ProfileBaseline – a forecast: the historical average headcount for each space, weekday and hour of day ("Tuesdays at 2 pm average 2.3 people").
  • FixedTimetable – a schedule with no forecast at all: HVAC on during fixed hours (default 8 am–8 pm, Monday–Friday, matching scheduled class hours), as many buildings run today.

 28class ProfileBaseline:
 29    """Average-profile forecast: mean headcount per space × weekday × hour.
 30
 31    If a combination was never seen in training, the forecast falls back to the
 32    space × hour average, then to the space average, so every hour gets a value.
 33
 34    Attributes:
 35        profile_: Learned means indexed by (space, weekday, hour_of_day).
 36        by_hour_: Fallback means indexed by (space, hour_of_day).
 37        by_space_: Fallback means indexed by space.
 38    """
 39
 40    keys = ["space", "weekday", "hour_of_day"]
 41
 42    def fit(self, train: pd.DataFrame, target: str = "mean_headcount") -> "ProfileBaseline":
 43        """Learn the average profile from training data.
 44
 45        Args:
 46            train: Hourly table with ``space``, ``weekday``, ``hour_of_day`` and
 47                the target column.
 48            target: Column to average (people).
 49
 50        Returns:
 51            The fitted model (to allow ``ProfileBaseline().fit(train)``).
 52        """
 53        self.profile_ = train.groupby(self.keys)[target].mean()
 54        self.by_hour_ = train.groupby(["space", "hour_of_day"])[target].mean()
 55        self.by_space_ = train.groupby("space")[target].mean()
 56        return self
 57
 58    def predict(self, df: pd.DataFrame) -> pd.Series:
 59        """Forecast mean headcount for each row of ``df``.
 60
 61        Args:
 62            df: Rows to forecast, with ``space``, ``weekday`` and ``hour_of_day``.
 63
 64        Returns:
 65            Forecast mean headcount (people), aligned with ``df``'s index.
 66
 67        Raises:
 68            AttributeError: If called before :meth:`fit`.
 69        """
 70        idx = pd.MultiIndex.from_frame(df[self.keys])
 71        pred = pd.Series(self.profile_.reindex(idx).to_numpy(), index=df.index)
 72        # Fall back to coarser averages where the exact combination is unseen.
 73        hour_idx = pd.MultiIndex.from_frame(df[["space", "hour_of_day"]])
 74        pred = pred.fillna(pd.Series(self.by_hour_.reindex(hour_idx).to_numpy(), index=df.index))
 75        pred = pred.fillna(df["space"].map(self.by_space_))
 76        return pred

 83class GradientBoostingForecaster:
 84    """Day-ahead machine-learning forecast using gradient-boosted decision trees.
 85
 86    Gradient boosting builds many small decision trees one after another; each
 87    new tree corrects the errors left by the trees before it. scikit-learn's
 88    ``HistGradientBoostingRegressor`` is used because it is fast on tens of
 89    thousands of rows, captures non-linear patterns and interactions (e.g.
 90    "2 pm *and* semester *and* Tuesday") without manual feature design, and
 91    accepts missing inputs (NaN) natively – important because sensor gaps
 92    leave some lag values unknown.
 93
 94    Inputs are the columns in :data:`occupancy.features.FEATURES`, all known at
 95    midnight before the forecast day. Forecasts are clipped at 0 people.
 96
 97    Args:
 98        learning_rate: How much each new tree is allowed to correct (smaller =
 99            slower, more cautious learning).
100        max_iter: Number of trees.
101        max_leaf_nodes: Size (complexity) of each tree.
102        min_samples_leaf: Minimum hours that must fall in each tree leaf; larger
103            values stop the model from memorising one-off events.
104        loss: ``"squared_error"`` or ``"poisson"`` (suited to non-negative counts).
105        random_state: Seed, so results are repeatable.
106        features: Input columns to use (default :data:`occupancy.features.FEATURES`).
107            Changed only for robustness tests, e.g. removing inputs.
108
109    Attributes:
110        model_: The fitted scikit-learn estimator.
111        spaces_: Space names seen in training (fixes the category coding).
112    """
113
114    def __init__(self, learning_rate: float = 0.05, max_iter: int = 300,
115                 max_leaf_nodes: int = 31, min_samples_leaf: int = 20,
116                 loss: str = "squared_error", random_state: int = 0,
117                 features: list[str] | None = None):
118        self.params = dict(learning_rate=learning_rate, max_iter=max_iter,
119                           max_leaf_nodes=max_leaf_nodes, min_samples_leaf=min_samples_leaf,
120                           loss=loss, random_state=random_state)
121        self.features = list(features) if features is not None else list(FEATURES)
122
123    def _matrix(self, df: pd.DataFrame) -> pd.DataFrame:
124        """Select the input columns and encode ``space`` as a fixed category."""
125        X = df[self.features].copy()
126        if "space" in X:
127            X["space"] = pd.Categorical(X["space"], categories=self.spaces_)
128        return X
129
130    def fit(self, train: pd.DataFrame, target: str = "mean_headcount") -> "GradientBoostingForecaster":
131        """Train on rows that have features (see :func:`occupancy.features.add_features`).
132
133        Args:
134            train: Training rows with every column in ``FEATURES`` and the target.
135            target: Column to forecast (people).
136
137        Returns:
138            The fitted model.
139        """
140        self.spaces_ = sorted(train["space"].unique())
141        self.model_ = HistGradientBoostingRegressor(categorical_features="from_dtype",
142                                                    early_stopping=False, **self.params)
143        self.model_.fit(self._matrix(train), train[target])
144        return self
145
146    def predict(self, df: pd.DataFrame) -> pd.Series:
147        """Forecast mean headcount (people, ≥ 0) for each row of ``df``."""
148        pred = np.clip(self.model_.predict(self._matrix(df)), 0, None)
149        return pd.Series(pred, index=df.index)
150
151    def input_importance(self, df: pd.DataFrame, target: str = "mean_headcount",
152                         n_repeats: int = 5) -> pd.DataFrame:
153        """Permutation importance: how much worse the forecast gets without each input.
154
155        Each input column is shuffled in turn (breaking its link to the
156        target) and the rise in MAE is recorded. A large rise means the model
157        relies on that input.
158
159        Args:
160            df: Rows to evaluate on (normally validation, not training, data).
161            target: Measured column.
162            n_repeats: Number of shuffles per input (results are averaged).
163
164        Returns:
165            Table with ``feature``, ``mae_increase`` (people) and ``std``.
166        """
167        res = permutation_importance(self.model_, self._matrix(df), df[target],
168                                     scoring="neg_mean_absolute_error",
169                                     n_repeats=n_repeats, random_state=0)
170        return pd.DataFrame({"feature": self.features, "mae_increase": res.importances_mean,
171                             "std": res.importances_std}).sort_values("mae_increase", ascending=False)

174class RandomForestForecaster(GradientBoostingForecaster):
175    """Alternative ML model: a random forest (used only as a comparison).
176
177    A random forest grows many deep decision trees *independently*, each on a
178    random resample of the training hours, and averages them. Unlike boosting,
179    the trees do not correct each other. Randomness means results vary
180    slightly with the seed. Takes the same inputs as
181    :class:`GradientBoostingForecaster`; ``space`` is coded as a number.
182
183    Args:
184        n_estimators: Number of trees.
185        min_samples_leaf: Minimum hours per leaf.
186        random_state: Seed for the random resampling.
187        features: Input columns (default :data:`occupancy.features.FEATURES`).
188    """
189
190    def __init__(self, n_estimators: int = 200, min_samples_leaf: int = 10,
191                 random_state: int = 0, features: list[str] | None = None):
192        self.params = dict(n_estimators=n_estimators, min_samples_leaf=min_samples_leaf,
193                           random_state=random_state, n_jobs=-1)
194        self.features = list(features) if features is not None else list(FEATURES)
195
196    def _matrix(self, df: pd.DataFrame) -> pd.DataFrame:
197        X = super()._matrix(df)
198        if "space" in X:
199            X["space"] = X["space"].cat.codes  # random forest needs numbers
200        return X
201
202    def fit(self, train: pd.DataFrame, target: str = "mean_headcount") -> "RandomForestForecaster":
203        """Train the forest (same interface as the gradient-boosting model)."""
204        self.spaces_ = sorted(train["space"].unique())
205        self.model_ = RandomForestRegressor(**self.params)
206        self.model_.fit(self._matrix(train), train[target])
207        return self

210class LinearRegressionForecaster:
211    """Alternative ML model: ordinary least-squares linear regression (comparison only).
212
213    The forecast is a weighted sum of the inputs. Space, hour of day and weekday
214    are one-hot encoded (one yes/no column per value), so each hour and weekday
215    gets its own fixed offset; but the model cannot learn *interactions* (e.g.
216    "2 pm matters more on semester weekdays") unless they are added by hand.
217    Missing recent-occupancy values are filled with the training mean, and an
218    extra yes/no column records that the value was missing. Forecasts are
219    clipped at 0 people.
220
221    Attributes:
222        model_: The fitted scikit-learn pipeline.
223    """
224
225    categorical = ["space", "hour_of_day", "weekday_num"]
226    numeric = ["in_semester", "lag_1d", "lag_7d", "mean_4w", "yesterday_mean"]
227
228    def fit(self, train: pd.DataFrame, target: str = "mean_headcount") -> "LinearRegressionForecaster":
229        """Fit the regression weights by least squares."""
230        encode = make_column_transformer(
231            (OneHotEncoder(handle_unknown="ignore"), self.categorical),
232            (SimpleImputer(strategy="mean", add_indicator=True), self.numeric))
233        self.model_ = make_pipeline(encode, LinearRegression())
234        self.model_.fit(train[self.categorical + self.numeric], train[target])
235        return self
236
237    def predict(self, df: pd.DataFrame) -> pd.Series:
238        """Forecast mean headcount (people, ≥ 0) for each row of ``df``."""
239        pred = self.model_.predict(df[self.categorical + self.numeric])
240        return pd.Series(np.clip(pred, 0, None), index=df.index)

243class FixedTimetable:
244    """Conventional HVAC timetable: on during fixed hours on chosen days.
245
246    This is an *assumed* schedule; the real Monash HVAC schedule was not
247    supplied with the data. The default 8 am–8 pm weekday window follows the
248    hours in which classes are scheduled (Monash's Student Timetable Procedure
249    permits teaching 8.00 am–9.00 pm Mon–Fri; weekday hours from 8 pm are occupied
250    less than half the time, and 8–8 performed best of the windows tested
251    on 2025 data).
252
253    Args:
254        start_hour: First hour of the day the system is on (inclusive).
255        end_hour: Hour at which the system turns off (exclusive).
256        weekdays_only: If True, the system is off all weekend.
257    """
258
259    def __init__(self, start_hour: int = 8, end_hour: int = 20, weekdays_only: bool = True):
260        self.start_hour = start_hour
261        self.end_hour = end_hour
262        self.weekdays_only = weekdays_only
263
264    def schedule(self, df: pd.DataFrame) -> pd.Series:
265        """Return True for each hour the timetable would condition the space.
266
267        Args:
268            df: Rows with ``hour_of_day`` and ``day_type`` columns.
269
270        Returns:
271            Boolean Series, True = HVAC on.
272        """
273        on = df["hour_of_day"].between(self.start_hour, self.end_hour - 1)
274        if self.weekdays_only:
275            on &= df["day_type"].eq("Weekday")
276        return on
